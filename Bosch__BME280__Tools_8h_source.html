48 unsigned int measurementCycleTime(bme280_settings& settings) {
49 // see Bosch Datasheet Appendix B: page 51
50 // max Measure time is defined as:
51 // T_Measure_MAX = 1.25
52 // + (2.3 * T_oversampling)
53 // + (2.3 * P_oversamplinmg + 0.575)
54 // + (2.3 * H_oversampling + 0.575)
55 float measureTime = 1.25f;
56 measureTime += 2.3f * getOversamplingFactor(settings.osr_t);
57 measureTime += 2.3f * getOversamplingFactor(settings.osr_p) + 0.575f;
58 measureTime += 2.3f * getOversamplingFactor(settings.osr_h) + 0.575f;
59 float standbyTime = getStandbyTime(settings.standby_time);
60
61 unsigned int measurementCycle = static_cast<unsigned int>(measureTime + standbyTime + 0.5f);
